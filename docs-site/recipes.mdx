## Diff, then validate what changed

Use the diff to find candidates, then the validator to test them on the trained model:

```python gpu
from circuitkit.diff import diff_graphs
from circuitkit.tracer import TracerBackend, nodes_for_units
from circuitkit.validate import GroupSpec, validate

report = diff_graphs(before, after)
units = [tuple(map(int, c.unit.split(":"))) for c in report.top]
backend = TracerBackend(trained_model)
nodes = nodes_for_units(backend, prompt, units, positions="all")
result = validate(backend, prompt, nodes, metric=metric, groups=[GroupSpec(name="top changes", nodes=nodes)])
```

## Save and reload any report

```python
from circuitkit.validate import Report, logit_diff, validate
from circuitkit.validate.toy import demo, demo_nodes

backend = demo()
report = validate(backend, "dallas", demo_nodes(backend),
                  metric=logit_diff(backend.token_id(" Austin"), backend.token_id(" Sacramento")))
text = report.model_dump_json(indent=1)
again = Report.model_validate_json(text)
assert again.groups[0].verdict == report.groups[0].verdict
```

## A custom metric

A metric is any function of the last position's logits:

```python
import numpy as np

from circuitkit.validate import Metric

keyword_ids, alternative = [101, 202, 303], 404
metric = Metric(
    "logsumexp(keywords) - logit(alternative)",
    lambda x: float(np.log(np.exp(x[keyword_ids]).sum()) - x[alternative]),
)
print(metric(np.zeros(500)))
```

## Make the HTML pages

```python
from circuitkit.report import render_validate, save_html
from circuitkit.validate import logit_diff, validate
from circuitkit.validate.toy import demo, demo_nodes

backend = demo()
report = validate(backend, "dallas", demo_nodes(backend),
                  metric=logit_diff(backend.token_id(" Austin"), backend.token_id(" Sacramento")))
save_html(render_validate(report), "validate.html")
```
